```rust
// tests/assert_e2e.rs

use assert_cmd::Command;
use predicates::prelude::*;
use std::process::{Child, Command as StdCommand};
use std::time::Duration;

#[test]
fn assert_passes_when_pin_matches() {
    let _sim = start_sim_blink();  // helper：起一个 blink demo
    std::thread::sleep(Duration::from_millis(500));
    Command::cargo_bin("moxin").unwrap()
        .args(&["assert", "pin D13 HIGH within 2s"])
        .assert()
        .success();
}

#[test]
fn assert_fails_with_exit_1_on_timeout() {
    let _sim = start_sim_blink();
    Command::cargo_bin("moxin").unwrap()
        .args(&["assert", "pin D7 HIGH within 100ms"])  // D7 没接
        .assert()
        .code(1);
}

#[test]
fn assert_serial_contains() { /* ... */ }

#[test]
fn assert_json_output() {
    Command::cargo_bin("moxin").unwrap()
        .args(&["assert", "--output", "json", "pin D13 HIGH within 100ms"])
        .assert()
        .stdout(predicate::str::contains("\"result\""));
}

// helper
fn start_sim_blink() -> Child { /* 起临时项目 + spawn moxin run */ }
```
