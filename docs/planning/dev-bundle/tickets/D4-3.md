### src/main.ino

```cpp
// 7-seg cathode 段 bitmap：bit0=A...bit6=G, bit7=DP
const byte DIGITS[10] = {
    0b00111111,  // 0
    0b00000110,  // 1
    0b01011011,  // 2
    0b01001111,  // 3
    0b01100110,  // 4
    0b01101101,  // 5
    0b01111101,  // 6
    0b00000111,  // 7
    0b01111111,  // 8
    0b01101111,  // 9
};

const int SEG_PINS[8] = {2, 3, 4, 5, 6, 7, 8, 9};   // A,B,C,D,E,F,G,DP

void setup() {
    for (int i = 0; i < 8; i++) pinMode(SEG_PINS[i], OUTPUT);
}

void show(byte bits) {
    for (int i = 0; i < 8; i++) {
        digitalWrite(SEG_PINS[i], (bits >> i) & 1);
    }
}

void loop() {
    for (int d = 0; d < 10; d++) {
        show(DIGITS[d]);
        delay(1000);
    }
}
```
