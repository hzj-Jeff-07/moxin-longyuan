### src/main.ino

```cpp
// 经典 do-re-mi 频率
const int NOTES[] = {262, 294, 330, 349, 392, 440, 494, 523};
const int N_NOTES = sizeof(NOTES) / sizeof(NOTES[0]);
const int NOTE_DURATION_MS = 500;

void setup() {
    pinMode(8, OUTPUT);
}

void loop() {
    for (int i = 0; i < N_NOTES; i++) {
        tone(8, NOTES[i]);
        delay(NOTE_DURATION_MS);
    }
    noTone(8);
    delay(500);
}
```
