### src/main.ino

```cpp
const int BTN_PIN = 2;
const int LED_PIN = 13;

void setup() {
    pinMode(BTN_PIN, INPUT_PULLUP);
    pinMode(LED_PIN, OUTPUT);
}

void loop() {
    int btn = digitalRead(BTN_PIN);
    // INPUT_PULLUP: 默认 HIGH，按下接 GND 变 LOW
    digitalWrite(LED_PIN, btn == LOW ? HIGH : LOW);
}
```
