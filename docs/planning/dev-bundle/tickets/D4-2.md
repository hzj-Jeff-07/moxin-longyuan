### 3. RunState 加数码管状态

```rust
#[derive(Debug, Clone, Default)]
pub struct SevenSegmentState {
    pub segments: u8,    // bit 0=A, bit 1=B, ..., bit 7=DP
}

impl SevenSegmentState {
    pub fn render(&self) -> [&'static str; 5] {
        let on = |bit: u8| self.segments & (1 << bit) != 0;
        // 返回 5 行字符串：上/上中/中/下中/下
        let top    = if on(0) { " ___ " } else { "     " };
        let upper_left  = if on(5) { "|" } else { " " };
        let upper_right = if on(1) { "|" } else { " " };
        let middle = if on(6) { " ___ " } else { "     " };
        let lower_left  = if on(4) { "|" } else { " " };
        let lower_right = if on(2) { "|" } else { " " };
        let bottom = if on(3) { " ‾‾‾ " } else { "     " };
        let dp = if on(7) { "." } else { " " };
        // ... 拼成 5 行（包含 DP）
        // 这是示意，实际拼装见代码实现
    }

    pub fn displayed_digit(&self) -> Option<char> {
        // 经典 7-seg 数字 → 段 bitmap
        match self.segments & 0x7F {  // 不含 DP
            0b0111111 => Some('0'),
            0b0000110 => Some('1'),
            0b1011011 => Some('2'),
            0b1001111 => Some('3'),
            0b1100110 => Some('4'),
            0b1101101 => Some('5'),
            0b1111101 => Some('6'),
            0b0000111 => Some('7'),
            0b1111111 => Some('8'),
            0b1101111 => Some('9'),
            0b1110111 => Some('A'),
            // ... 其他字符
            _ => None,
        }
    }
}

pub struct RunState {
    // ...
    pub seven_segs: HashMap<String, SevenSegmentState>,    // key = 实例 id
}
```

### 4. apply_event 触发数码管段更新

```rust
// 在 Pin 事件处理里
if let Some((seg_id, bit_idx)) = pin_to_seg.get(&d) {
    let seg_state = s.seven_segs.entry(seg_id.clone()).or_default();
    if value != 0 {
        seg_state.segments |= 1 << bit_idx;
    } else {
        seg_state.segments &= !(1 << bit_idx);
    }
}
```

`pin_to_seg: HashMap<u8, (String, u8)>` 索引由 shell 在加载 project 时构建。

## 单元测试

```rust
#[test]
fn seven_segment_digit_0() {
    let s = SevenSegmentState { segments: 0b0111111 };
    assert_eq!(s.displayed_digit(), Some('0'));
}

#[test]
fn seven_segment_digit_8() {
    let s = SevenSegmentState { segments: 0b1111111 };
    assert_eq!(s.displayed_digit(), Some('8'));
}

#[test]
fn seven_segment_blank() {
    let s = SevenSegmentState { segments: 0 };
    assert_eq!(s.displayed_digit(), None);
}

#[test]
fn seven_segment_render_has_5_lines() {
    let s = SevenSegmentState { segments: 0b0111111 };
    assert_eq!(s.render().len(), 5);
}
```
